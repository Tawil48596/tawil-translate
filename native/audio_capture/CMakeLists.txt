cmake_minimum_required(VERSION 3.20)
project(tawil_audio_capture LANGUAGES CXX)
add_executable(tawil-audio-capture main.cpp)
target_compile_features(tawil-audio-capture PRIVATE cxx_std_20)
target_compile_definitions(tawil-audio-capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(tawil-audio-capture PRIVATE ole32 mmdevapi uuid)
if(MSVC)
  target_compile_options(tawil-audio-capture PRIVATE /W4 /permissive- /EHsc)
endif()
install(TARGETS tawil-audio-capture RUNTIME DESTINATION bin)
